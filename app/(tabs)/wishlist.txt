import React from "react";
import {
  View,
  Text,
  FlatList,
  TouchableOpacity,
  ActivityIndicator,
} from "react-native";
import { useQuery } from "@tanstack/react-query";
import { fetchStockQuotes } from "../api/marketApi";
import { useWatchlistStore } from "../store/watchlistStore";
import { Ionicons } from "@expo/vector-icons";
import { Link } from "expo-router";

const WishlistScreen = () => {
  const { stocks, removeStock } = useWatchlistStore();

  const { data: quotes, isLoading } = useQuery({
    queryKey: ["stock-quotes", stocks],
    queryFn: () => fetchStockQuotes(stocks),
    enabled: stocks.length > 0,
  });

  const renderItem = ({ item, index }: { item: any; index: number }) => {
    const symbol = stocks[index];
    const quote = item?.body?.[0];

    if (!quote) return null;

    const priceChange = quote.regularMarketChangePercent;
    const isPositive = priceChange >= 0;

    return (
      <Link href={`/stock/${symbol}`} asChild>
        <TouchableOpacity className="flex-row items-center p-4 border-b border-gray-200">
          <View className="flex-1">
            <Text className="text-lg font-semibold">{symbol}</Text>
            <Text className="text-gray-600">
              ${quote.regularMarketPrice?.toFixed(2)}
            </Text>
          </View>
          <View className="items-end">
            <Text
              className={`text-sm ${
                isPositive ? "text-green-500" : "text-red-500"
              }`}
            >
              {isPositive ? "+" : ""}
              {priceChange?.toFixed(2)}%
            </Text>
            <TouchableOpacity
              className="p-2"
              onPress={() => removeStock(symbol)}
            >
              <Ionicons name="trash-outline" size={20} color="#666" />
            </TouchableOpacity>
          </View>
        </TouchableOpacity>
      </Link>
    );
  };

  if (stocks.length === 0) {
    return (
      <View className="flex-1 justify-center items-center bg-white">
        <Ionicons name="star-outline" size={64} color="#666" />
        <Text className="text-gray-500 text-lg mt-4">
          Your watchlist is empty
        </Text>
        <Text className="text-gray-400 text-center mt-2 px-8">
          Add stocks to your watchlist to track their performance
        </Text>
      </View>
    );
  }

  return (
    <View className="flex-1 bg-white">
      {isLoading ? (
        <View className="flex-1 justify-center items-center">
          <ActivityIndicator size="large" color="#000" />
        </View>
      ) : (
        <FlatList
          data={stocks}
          renderItem={renderItem}
          keyExtractor={(item) => item}
          className="flex-1"
        />
      )}
    </View>
  );
};

export default WishlistScreen;
