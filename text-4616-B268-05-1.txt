// api/marketApi.ts
import axios from 'axios';

const API_KEY = '9288c8ca29msh202e3f882223132p1ed9ebjsna02b8d7935f6';
const API_HOST = 'yahoo-finance15.p.rapidapi.com';

interface MarketTickerResponse {
  meta: {
    version: string;
    status: number;
    copywrite: string;
    totalrecords: number;
    headers: {
      symbol: string;
      name: string;
      lastsale: string;
      netchange: string;
      pctchange: string;
      marketCap: string;
    };
  };
  body: StockTicker[];
}

export interface StockTicker {
  symbol: string;
  name: string;
  lastsale: string;
  netchange: string;
  pctchange: string;
  marketCap: string;
}

export async function fetchMarketTickers(page: number = 1, type: string = 'STOCKS'): Promise<MarketTickerResponse> {
  const options = {
    method: 'GET',
    url: 'https://yahoo-finance15.p.rapidapi.com/api/v2/markets/tickers',
    params: {
      page: page.toString(),
      type
    },
    headers: {
      'x-rapidapi-key': API_KEY,
      'x-rapidapi-host': API_HOST
    }
  };

  try {
    const response = await axios.request(options);
    return response.data;
  } catch (error) {
    console.error('Error fetching market tickers:', error);
    throw error;
  }
}

// components/StockCard.tsx
import React from 'react';
import { View, Text } from 'react-native';
import { StockTicker } from '../api/marketApi';

interface StockCardProps {
  stockData: StockTicker;
}

export function StockCard({ stockData }: StockCardProps) {
  const isPositive = !stockData.netchange.startsWith('-');
  
  return (
    <View className="bg-white rounded-lg p-4 shadow-sm">
      <View className="flex-row justify-between items-center">
        <View className="flex-1">
          <Text className="text-lg font-bold">{stockData.symbol}</Text>
          <Text className="text-sm text-gray-500" numberOfLines={1}>{stockData.name}</Text>
        </View>
        
        <View className="items-end">
          <Text className="text-lg font-semibold">{stockData.lastsale}</Text>
          <View className="flex-row items-center">
            <Text 
              className={`text-sm ${isPositive ? 'text-green-600' : 'text-red-600'}`}
            >
              {stockData.netchange} ({stockData.pctchange})
            </Text>
          </View>
        </View>
      </View>
    </View>
  );
}
